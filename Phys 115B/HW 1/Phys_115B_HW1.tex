\textbf{You just know the answer}

A common problem in QM is ``find the energies of the following Hamiltonian.'' The idea
is to do as little math as possible but to build intuition. Describe the lowest several
eigenenergies of the following Hamiltonians. It is up to you to decide what value of ``several''
best illustrates what is going on. Answers are short.

\textbf{Example:} What are the energies of the two separate simple harmonic oscillators, each
with frequency $\omega$?

\textbf{Answer:} The oscillators have energies
\[
E_1 = \hbar\omega(1/2 + n_1), \quad n_1 = [0,1,2,3,\ldots]
\]
\[
E_2 = \hbar\omega(1/2 + n_2), \quad n_2 = [0,1,2,3,\ldots]
\]
so the combined oscillators have energy
\[
E = E_1 + E_2 = \hbar\omega(1 + n_1 + n_2)
= \hbar\omega[1,2,2,3,3,3,4,4,4,4,\ldots]
\]

\begin{enumerate}
\item Particle of mass $m$ in one dimensional box of width $a$.
\item Particle of mass $m$ in two dimensional box of width $a$ and length $b$.
\item Two disconnected particles, each of mass $m$, one in a 1D box of width $a$ and the
other in a 1D box of width $b$.
\item Particle of mass $m$ confined to a circular ring of circumference $a$.
\item Particle of mass $m$ in a one dimensional potential $V = \frac{1}{2}m\omega^2 x^2$.
\item Paricle of mass $m$ in a two dimensional potential
$V = \frac{1}{2}m\omega^2(x^2 + y^2)$.
\end{enumerate}

\begin{ques}\label{q4}
\textbf{1D to 3D} (Griffiths 4.1, same in both editions)

\begin{itemize}
    \item[(a)] Work out all of the \textbf{canonical commutation relations} for components of the operators $\br$ and $\bp$: $[\hat{x},\hat{y}], [\hat{x},\hat{p}_y], [\hat{x},\hat{p}_x], [\hat{p}_y,\hat{p}_z]$, and so on.
    \item[(b)] Confirm the three-dimensional version of \textbf{Ehrenfest's Theorem},
    $$\frac{d}{dt}\left<\br\right> = \frac{1}{m}\left<\bp\right>,\quad \frac{d}{dt}\left<\bp\right> = \left<-\nabla V\right>$$
    \item[(c)] Formulate \textbf{Heisenberg's Uncertainty Principle} in three dimensions.
\end{itemize}
\end{ques}

\begin{proof}
    
    \hfil

    \begin{itemize}
        \item[(a)] Given $j,k \in \{x,y,z\}$, the action of $[\hat{j},\hat{k}]$ on the states in position basis are as follow:
        \begin{align}
            [\hat{j},\hat{k}]\ket{\psi}&= (\hat{j}\hat{k}-\hat{k}\hat{j})\int_{\RR^3}\psi(\hat{r})\ket{\bar{r}}dr^3 = (\hat{jk}\int_{\RR^3}k\psi(\bar{r})\ket{\bar{r}}dr^3 - \hat{k}\int_{\RR^3}j\psi(\bar{r})\ket{\bar{r}}dr^3)\\
            &= \int_{\RR^3}jk\psi(\bar{r})\ket{\bar{r}}dr^3 - \int_{\RR^3}kj\psi(\bar{r})\ket{\bar{r}}dr^3 = 0
        \end{align}

        Similarly, the action of $[\hat{p}_j, \hat{p}_k]$ on the states as follow:
        \begin{align}
            [\hat{p}_j,\hat{p}_k]\ket{\psi} &= -\hbar^2\left(\frac{\partial}{\partial j}\frac{\partial}{\partial k}-\frac{\partial}{\partial k}\frac{\partial}{\partial j}\right)\int_{\RR^3}\psi(\bar{r})\ket{\bar{r}}dr^3\\
            &= -\hbar^2\left(\frac{\partial}{\partial j}\int_{\RR^3}\frac{\partial \psi}{\partial k}\ket{\bar{r}}dr^3 - \frac{\partial}{\partial k}\int_{\RR^3}\frac{\partial \psi}{\partial j}\ket{\bar{r}}dr^3\right)\\
            &= -\hbar^2\left(\int_{\RR^3}\frac{\partial}{\partial j}\left(\frac{\partial \psi}{\partial k}\right)\ket{\bar{r}}dr^3 - \int_{\RR^3}\frac{\partial}{\partial k}\left(\frac{\partial \psi}{\partial j}\right)\ket{\bar{r}}dr^3\right) = 0
        \end{align}
        Which, it's due to the fact that the mixed partial derivatives are equal if the function is smooth / at least twice continuously differentiable.

        Now, the action of $[\hat{j}, \hat{p}_k]$ is as follow:
        \begin{align}
            [\hat{j},\hat{p}_k]\ket{\psi} &= -i\hbar\left(\hat{j}\frac{\partial}{\partial k}-\frac{\partial}{\partial k}\hat{j}\right)\int_{\RR^3}\psi(\bar{r})\ket{\bar{r}}dr^3\\
            &= -i\hbar\left(\hat{j}\int_{\RR^3}\frac{\partial\psi}{\partial k}\ket{\bar{r}}dr^3 - \frac{\partial}{\partial k}\int_{\RR^3}j\psi(\bar{r})\ket{\bar{r}}dr^3\right)\\
            &= -i\hbar\left(\int_{\RR^3}j\frac{\partial\psi}{\partial k}\ket{\bar{r}}dr^3 - \int_{\RR^3}\left(\frac{\partial j}{\partial k}\psi(\bar{r}) + j \frac{\partial \psi}{\partial k}\right)\ket{\bar{r}}dr^3\right)\\
            &= -i\hbar\left(\int_{\RR^3}j\frac{\partial\psi}{\partial k}\ket{\bar{r}}dr^3  - \delta_{j k}\int_{\RR^3}\psi(\bar{r})\ket{\bar{r}}dr^3 - \int_{\RR^3}j\frac{\partial\psi}{\partial k}\ket{\bar{r}}dr^3\right)\\
            &= i\hbar \delta_{j k}\ket{\psi}
        \end{align}
        So, as a result, we get:
        \begin{align}
            [\hat{x},\hat{y}] = 0,\quad [\hat{x},\hat{p}_y] = 0,\quad [\hat{x},\hat{p}_x] = i\hbar,\quad [\hat{p}_y,\hat{p}_z] = 0
        \end{align}

        \hfil

        \item[(b)] If limit to one dimension (pick $j\in\{x,y,z\}$), then Ehrenfest's theorem in one dimension guarantees that $\frac{d}{dt}\left<j\right> = \frac{1}{m}\left<p_j\right>$ and $\frac{d}{dt}\left<p_j\right> = \left<-\frac{\partial V}{\partial j}\right>$. Hence, the three-dimensional version follows, since now if $\bp$ and $\br$ are $(\hat{p}_x,\hat{p}_y,\hat{p}_z)$ and $(\hat{x},\hat{y},\hat{z})$ respectively, one has:
        \begin{align}
            \frac{d}{dt}\left<\br\right> &= \frac{d}{dt}(\left<x\right>,\left<y\right>,\left<z\right>) = \frac{1}{m}(\left<p_x\right>,\left<p_y\right>,\left<p_z\right>) = \frac{1}{m}\left<\bp\right>
        \end{align}
        \begin{align}
            \frac{d}{dt}\left<\bp\right> &= \frac{d}{dt}(\left<p_x\right>,\left<p_y\right>,\left<p_z\right>) = \left(\left<-\frac{\partial V}{\partial x}\right>, \left<-\frac{\partial V}{\partial y}\right>, \left<-\frac{\partial V}{\partial z}\right>\right) = \left<-\nabla V\right>
        \end{align}
        Since $-\nabla V = \left(-\frac{\partial V}{\partial x},-\frac{\partial V}{\partial y},-\frac{\partial V}{\partial z}\right)$.

        \hfil
    
        \item[(c)] Given $j,k \in \{x,y,z\}$, since $[\hat{j},\hat{p}_k] = i\hbar \delta_{j k}$ (verified in part (a)), then by the generalized uncertainty principle, one has:
        \begin{align}
            \sigma_j \sigma_{p_k} \geq \frac{1}{2}\left|[\hat{j},\hat{p}_k]\right| = \frac{\hbar}{2}\delta_{j k}
        \end{align}
        Hence, if $j=k$, then $\sigma_j \sigma_(p_j)\geq \frac{\hbar}{2}$, yet if $j\neq k$, one has $\sigma_j\sigma_{p_k}\geq 0$, which imposes no restriction.
    \end{itemize}
\end{proof}

\begin{ques}\label{q6}
\textbf{Practice with Spherical Harmonics} (Griffiths 4.4 in 3rd edition, 4.3 in 2nd edition)

Show that 
$$\Theta(\theta) = A \ln[\tan(\theta/2)]$$
satisfies the $\Theta$ equation for $\ell=m=0$. This is the unacceptable "second solution" - what's wrong with it?

Note: the equation is of the form 
$$\sin(\theta)\frac{d}{d\theta}\left(\sin(\theta)\frac{d\Theta}{d\theta}\right)+[\ell(\ell+1)\sin^2(\theta)-m^2]\Theta = 0$$
\end{ques}

\begin{proof}
    Given $\ell=m=0$, the given equation reduces to:
    \begin{align}
        \sin(\theta)\frac{d}{d\theta}\left(\sin(\theta)\frac{d\Theta}{d\theta}\right)=0
    \end{align}
    Hence, one gets $\sin(\theta)\frac{d\Theta}{d\theta} = A_1$ for some constant $A_1 \in \CC$.

    Now, since this implies $\frac{d\Theta}{d\theta} = \frac{A_1}{\sin(\theta)} = A_1 \csc(\theta)$, then one gets the following:
    \begin{align}
        \Theta &= \int A_1 \csc(\theta) d\theta = -A_1 \ln\left|\tan\left(\frac{\theta}{2}\right)\right|+A_2
    \end{align}
    Which, for $\theta\in (0,\pi)$, $\tan(\theta/2)>0$, hence $\Theta = A \ln[\tan(\theta/2)]+B$ for some $A,B \in \CC$.

    However, this is not an acceptable solution, since the integration of $|\Theta|^2$ over $\theta\in (0,\pi)$ in fact diverges, showing that $\Theta$ is not normalizable.
\end{proof}
