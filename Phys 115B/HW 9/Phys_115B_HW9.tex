\section*{1 D}
\begin{ques}\label{q1}
Consider the parity operator $\hat{\Pi}$ in three dimensions, such that $\hat{\Pi}\psi(\mathbf r)=\psi(-\mathbf r)$.

\begin{itemize}
\item[(a)] Show that for states $\psi$ expressed in polar coordinates, the action of the parity operator can be written as
\[
\hat{\Pi}\psi(r,\theta,\phi)=\psi(r,\pi-\theta,\phi+\pi)
\]

\item[(b)] Show that for hydrogenic energy eigenstates $\psi_{nlm}(r,\theta,\phi)$ we have
\[
\hat{\Pi}\psi_{nlm}=(-1)^l\psi_{nlm}(r,\theta,\phi)
\]
\end{itemize}
\end{ques}

\begin{proof}
    
    \hfil

    \begin{itemize}
        \item[(a)] Given the parity operator has the coordinate change $\br \mapsto -\br$, one has the following on each component:
        \begin{align}
            &x: \quad r\sin(\theta)\cos(phi) \mapsto -r\sin(\theta)\cos(\phi) = r\sin(\pi-\theta)\cos(\phi+\pi)\\
            &y: \quad r\sin(\theta)\sin(phi) \mapsto -r\sin(\theta)\sin(\phi) = -r\sin(\pi-\theta)\sin(\phi) = r\sin(\theta)\sin(\phi+\pi)\\
            &z:\quad r\cos(theta) \mapsto -r\cos(\theta) = r\cos(\pi-\theta)
        \end{align}
        So, one transformation on the spherical coordinates that's compatible with the parity operator is $r\mapsto r$, $\theta\mapsto \pi-\theta$, and $\phi\mapsto \phi+\pi$.

        Hence, using $r,\theta,\psi$ as the input, one has:
        \begin{align}
            \hat{\Pi}\psi(r,\theta,\phi) = \psi(r,\pi-\theta, \phi+\pi)
        \end{align}

        \hfil

        \item[(b)] Since for $\psi_{nlm} = R_{n,l} (r) Y^m_l (\theta,\phi)$ in general, then the only one affected by the parity is $Y^m_l(\theta,\phi)$. 
        
        Recall the formula $Y^m_l(\theta,\phi) = e^{im\phi}P^m_l (\cos(\theta))$, where $P^m_l$ is given as follow:
        \begin{align}
            P^m_l (z)=(1-z^2)^{|m|/2}\left(\frac{d}{dz}\right)^{|m|}P_l (z), \quad P_l (z) := \frac{1}{2^l l!}\left(\frac{d}{dz}\right)^l (z^2-1)^l
        \end{align}
        Which, it can be shown inductively, that all $P_l (z)$ only contains the monomials with the degree of same parity with $l$; then, one can also inductively shown $\left(\frac{d}{dz}\right)^{|m|}P_l (z)$ has all monomial with degree of same parity with $l-|m|$, showing $P^m_l (-z) = (-1)^{l-|m|}P^m_l (z)$. Finally, one has $e^{im(\phi+\pi)} = (-1)^{|m|}e^{im\phi}$. Hence, with $\cos(\pi-\theta) = -\cos(\theta)$, one has the following:
        \begin{align}
            \hat{\Pi}\psi_{nlm} &= \psi_{nlm}(r,\pi-\theta, \phi+\pi) = R_{n,l}(r)e^{im(\phi+\pi)}P^m_l (\cos(\pi-\theta))\\
            &= R_{n,l}(r)(-1)^{|m|}e^{im\phi}P^m_l (-cos(\theta))= R_{n,l} (-1)^{|m|}e^{im\phi}(-1)^{l-|m|}P^m_l (\cos(\theta))\\
            &= (-1)^l R_{n,l}e^{im\phi}P^m_l (\cos(\theta))= (-1)^l\psi_{nlm}(r,\theta,\phi)
        \end{align} 
    \end{itemize}
\end{proof}

\begin{ques}\label{q2}
We can classify operators depending on how they transform under parity, $\hat{Q}\rightarrow \hat{\Pi}^\dagger \hat{Q}\hat{\Pi}$ (see example 6.2 or Table 6.1 in the book can be useful).

\begin{itemize}
\item[(a)] In one dimension, show that position and momentum are odd under parity.

\item[(b)] In three dimensions, operators that are invariant under parity and commute with $\hat L$ are called scalar operators, $\hat{\Pi}^\dagger \hat Q_s \hat{\Pi}=\hat Q_s$. Operators that flip sign under parity and commute with $\hat L$ are called pseudo-scalar operators, $\hat{\Pi}^\dagger \hat Q_s \hat{\Pi}=-\hat Q_s$. Show that scalar operators satisfy $[\hat{\Pi},\hat Q_s]=0$, while pseudoscalar operators satisfy
\[
\{\hat{\Pi},\hat Q_s\}=0
\]
(where the curly brackets denote the anti-commutator).

\item[(c)] Operators that flip sign under parity and whose commutators with $\hat \bL$ are like those of $\hat \bL$ are called vector operators, such that $\hat{\Pi}^\dagger \hat Q_v \hat{\Pi}=-\hat Q_v$. Operators that are invariant under parity and whose commutators with $\hat \bL$ are like those of $\hat \bL$ are called pseudovector or axial vector operators, $\hat{\Pi}^\dagger \hat Q_a \hat{\Pi}=\hat Q_a$. Show that vector operators satisfy
\[
\{\hat{\Pi},\hat Q_v\}=0
\]
and pseudo-vectors satisfy
\[
[\hat{\Pi},\hat Q_a]=0.
\]

\item[(d)] Give an example of a scalar operator, pseudoscalar operator, a vector operator and an axial or pseudo vector operator (feel free to Google or use the book).
\end{itemize}
\end{ques}

\begin{proof}
    Here, it uses the fact that $\hat{\Pi}^\dagger=\hat{\Pi}^{-1}=\hat{\Pi}$.

    \hfil

    \begin{itemize}
        \item[(a)] Let's first recall some tools, for instance $\braket{x}{p}=e^{ipx/\hbar}$. Then, we have the following:
        \begin{align}
            \hat{Pi}^\dagger \hat{x} \hat{\Pi} \ket{x} = \hat{\Pi}\hat{x}\ket{-x} = -x\hat{Pi}\ket{-x} = -x\ket{x}
        \end{align}
        Since above is true for all $x\in\RR$, it shows $\hat{\Pi}^\dagger\hat{x}\hat{\Pi}=-\hat{x}$.

        Similarly, we have the following:
        \begin{align}
            \hat{\Pi}\ket{p} &= \int \hat{\Pi}\ket{x}\braket{x}{p}dx = \int e^{ipx/\hbar}\ket{-x}dx\\
            &= \int e^{i(-p)u/\hbar}\ket{u}du = \ket{-p}
        \end{align}
        (Note: Here $u:=-x$).

        Hence, the full relation on momentum is:
        \begin{align}
            \hat{\Pi}^\dagger\hat{p}\hat{\Pi}\ket{p} = \hat{\Pi}\hat{p}\ket{-p} = -p\hat{\Pi}\ket{-p} = -p\ket{p}
        \end{align}
        Since above is true for all $p \in\RR$, it shows $\hat{\Pi}^\dagger\hat{p}\hat{\Pi}=-\hat{p}$.

        \hfil

        \item[(b)] For scalar operators we have the following:
        \begin{align}
            \hat{\Pi}^\dagger \hat{Q}_s \hat{\Pi} = \hat{Q}_s \implies \hat{Q}_s \hat{\Pi} = \hat{\Pi}\hat{Q}_s \implies [\hat{\Pi},\hat{Q}_s]=0
        \end{align}
        For pseudo-scalar operators we have the following:
        \begin{align}
            \hat{\Pi}^\dagger\hat{Q}_s\hat{\Pi}=-\hat{Q}_s \implies \hat{Q}_s \hat{\Pi}=-\hat{\Pi}\hat{Q}_s \implies \{\hat{\Pi},\hat{Q}_s\}=0
        \end{align}

        \hfil

        \item[(c)] Similarly to the previous part, for vector operators, we have:
        \begin{align}
            \hat{\Pi}^\dagger\hat{Q}_v\hat{\Pi}=-\hat{Q}_v\implies \hat{Q}_v\hat{\Pi}=-\hat{\Pi}\hat{Q}_v \implies {\hat{\Pi},\hat{Q}_v}=0
        \end{align}
        For pseudovector operators, we have:
        \begin{align}
            \hat{\Pi}^\dagger\hat{Q}_a\hat{\Pi} = \hat{Q}_a \implies \hat{Q}_a\hat{\Pi}=\hat{\Pi}\hat{Q}_a \implies [\hat{\Pi},\hat{Q}_a]=0
        \end{align}

        \hfil

        \item[(d)] the free particle hamiltonian $H = \frac{1}{2m}(p_x^2+p_y^2+p_z^2)$ is a scalar operator (since all momentum operators $p_u$ is odd under parity, so $\hat{\Pi}^\dagger p_u^2 \hat{\Pi} =(\hat{\Pi}^\dagger p_u \hat{\Pi})(\hat{\Pi}^\dagger p_u \hat{\Pi}) = (-p_u)^2 = p_u^2$).
        
        In three dimension, the operator $x$ is a pseudoscalar operator (since it's proven to have odd parity under $1$-dimension, here it's still true).

        In three dimension, the operator $\br$ is a vector operator, since it's odd under parity and its transformation with $\bL$ is some variation of itself.

        In three dimension, the operator $\bL$ is a pseudovector operator, since it doesn't change under parity (because for any distinct $u,v,t \in \{x,y,z\}$, the component $L_u = v p_t - t p_v$ up to some sign, which must be even since its summands are products of two odds), while transormation with itself is also a variation of itself.
    \end{itemize}
\end{proof}

\section*{3 D}
\begin{ques}\label{q3}
Consider a particle of mass $m$ and charge $q$ confined to a circular ring in the $x$-$y$ plane, with radius $r$. The system will be driven by an electric field $E_x$ in the $x$ direction (which can stimulate photon induced transitions). It is natural to work in polar coordinates for this problem, with $x=r\cos\theta$.

\begin{itemize}
\item[(a)] What are the first five energies of the system with $E_x=0$?

\item[(b)] What are the first five eigenfunctions $\psi(\theta)$? Please normalize them, and pick your basis such that the eigenfunctions are also eigenfunctions of $L_z$.

\item[(c)] Explicitly write out a $3\times3$ matrix for the Hamiltonian
\[
H' = E_x q x = E q r \cos\theta
\]
This matrix will use the three lowest states above. Extrapolate this result to write a concise selection rule that tells what transitions are allowed for any state.

\item[(d)] What frequencies are absorbed by a “gas” of such rings if they are cold enough to be initially in the ground state?

\item[(e)] What additional frequencies are absorbed by rings initially in the first excited state?
\end{itemize}
\end{ques}

\begin{proof}
    Before starting, given the electric potential $\varphi$, one has $-\nabla \varphi = (E_x,0)$, showing that $\varphi = -E_x x+C$ for some constant $C\in\RR$. For simplicity, choose $C=0$, hence $\varphi= E_x x = -E_x r\cos(\theta)$.

    \hfil

    \begin{itemize}
        \item[(a)] Suppose $E_x=0$, then the system's hamiltonian is simply $H=-\frac{p^2}{2m}$. By the restriction to $xy$-plane, $p^2=-\hbar^2\nabla^2$, with the restriction that only $\theta$ varies while $r$ is fixed, so one has $\nabla^2 = \frac{1}{r}\frac{\partial}{\partial r}\left(r\frac{\partial}{\partial r}\right)+\frac{1}{r^2}\frac{\partial^2}{\partial \theta^2}=\frac{1}{r^2}\frac{\partial^2}{\partial \theta^2}$.
        
        Hence, any eigenfunction $\psi$ with energy $E\geq 0$ satisfies the following equation:
        \begin{align}
            E\psi = H\psi = -\frac{\hbar^2}{2mr^2}\frac{\partial^2}{\partial\theta^2}\psi\ \implies\ \frac{\partial^2\psi}{\partial \theta^2}+\left(\frac{r\sqrt{2mE}}{\hbar}\right)^2\psi=0
        \end{align}
        Then, with $w := \frac{r\sqrt{2mE}}{\hbar}$, this implies the solutions are $\psi(\theta) = Ke^{iw\theta}+Le^{-iw\theta}$; with the restriction that it's a ring (or $\psi(\theta)=\psi(\theta+2\pi)$), one must have $w\in \ZZ$, showing $\frac{r\sqrt{2mE}}{\hbar}=n$ for some $n\in\ZZ$, or $E = \frac{\hbar^2n^2}{2mr^2}$.

        Hence, the first five lowest energies are:
        \begin{align}
            &n=0\implies E_0 = 0,\quad n=\pm 1\implies E_1=\frac{\hbar^2}{2mr^2},\quad n=\pm 2\implies E_2=\frac{2\hbar^2}{mr^2}
        \end{align}

        \hfil

        \item[(b)] Given that the angular momentum $L_z = -i\hbar\frac{\partial}{\partial\theta}$ (here $\theta$ is the polar angle), then the eigenfunctions must be of the form $Ke^a\theta$ for some $K,a \in\CC$. Together with the first part, which has the following eigenfunctions:
        \begin{align}
            \forall n\in\ZZ,\quad w=n, \quad \psi_n = K_n e^{in\theta}
        \end{align}
        To normalize it, we need the following ondition to hold:
        \begin{align}
            1=\int_{0}^{2\pi}|\psi_n|^2d\theta = \int_{0}^{2\pi}|K_n|^2 d\theta = 2\pi|K_n|^2
        \end{align}
        This implies that $|K_n| = \frac{1}{\sqrt{2\pi}}$. For simplicity, choose $K_n>0$, one has the following eigenfunctions after normalizing:
        \begin{align}
            &n=0\implies \psi_0 = \frac{1}{\sqrt{2\pi}},\quad n=1\implies \psi_1=\frac{1}{\sqrt{2\pi}}e^{i\theta},\quad n=-1\implies \psi_{-1}=\frac{1}{\sqrt{2\pi}}e^{-i\theta}\\
            &n=2\implies \psi_2=\frac{1}{\sqrt{2\pi}}e^{i2\theta},\quad n=-2\implies \psi_{-2}=\frac{1}{\sqrt{2\pi}}e^{-i2\theta}
        \end{align}

        \hfil

        \item[(c)] Given the hamiltonian $H' = Eqr\cos(\theta)$, with the three lowest energy states $n=0,1,-1$ in order for the basis. Then, the transformation $\psi \mapsto \cos(\theta)\psi$ can be rewritten as follow for each $n$:
        \begin{align}
            &n=0,\quad \psi_0 = \frac{1}{\sqrt{2\pi}}\mapsto \frac{1}{\sqrt{2\pi}}\cos(\theta)\\
            &n=1,\quad \psi_1 = \frac{1}{\sqrt{2\pi}}e^{i\theta}\mapsto \frac{1}{\sqrt{2\pi}}\cos(\theta)e^{i\theta}\\
            &n=-1,\quad \psi_{-1} = \frac{1}{\sqrt{2\pi}}e^{-i\theta}\mapsto \frac{1}{\sqrt{2\pi}}\cos(\theta)e^{-i\theta}
        \end{align}
        Let's do the Fourier Decomposition of the three:
        \begin{itemize}
            \item For $n=0$, based on the fact that $e^{in\theta}=\cos(n\theta)+i\sin(n\theta)$, we have the following:
            \begin{align}
                \psi_1+\psi_{-1} = \frac{1}{\sqrt{2\pi}}(e^{i\theta}+e^{-i\theta})=\frac{1}{\sqrt{2\pi}}2\cos(\theta)\ \implies\ \frac{1}{\sqrt{2\pi}}\cos(\theta)=\frac{1}{2}\psi_1+\frac{1}{2}\psi_{-1}
            \end{align}

            \item For $n=1$, one requires integral:
            \begin{align}
                c_0 &= \int_{0}^{2\pi}\frac{1}{\sqrt{2\pi}}\cos(\theta)e^{i\theta}\frac{1}{\sqrt{2\pi}}d\theta = \frac{1}{2\pi}\left(\int_{0}^{2\pi}\cos^2(\theta)d\theta+i\int_{0}^{2\pi}\sin(\theta)\cos(\theta)d\theta\right)\\
                &= \frac{1}{2\pi}\left(\int_{0}^{2\pi}\frac{1+\cos(2\theta)}{2}d\theta+\frac{i}{2}\int_{0}^{2\pi}\sin(2\theta)d\theta\right)\\
                &= \frac{1}{2\pi}\cdot \frac{2\pi}{2} = \frac{1}{2}
            \end{align}
            \begin{align}
                c_1 = \int_{0}^{2\pi}\frac{1}{\sqrt{2\pi}}\cos(\theta)e^{i\theta}\frac{1}{\sqrt{2\pi}}e^{-i\theta}d\theta = \frac{1}{2\pi}\int_{0}^{2\pi}\cos(\theta)d\theta=0
            \end{align}
            \begin{align}
                c_{-1} &= \int_{0}^{2\pi}\frac{1}{\sqrt{2\pi}}\cos(\theta)e^{i\theta}\frac{1}{\sqrt{2\pi}}e^{i\theta}d\theta = \frac{1}{2\pi}\int_{0}^{2\pi}\cos(\theta)e^{i2\theta}d\theta\\
                &= \frac{1}{2\pi}\left(\int_{0}^{2\pi}\cos(\theta)\cos(2\theta)d\theta+i\int_{0}^{2\pi}\cos(\theta)\sin(2\theta)d\theta\right)\\
                &= 0
            \end{align}
            
            \item For $n=-1$, notice that the imaginary part of $d_0$ is the negative of $c_0$ for the case $n=1$, hence $d_0=-1$; Similarly, $d_1$ is the complex conjugate of $c_{-1}$ above, hence $d_1=0$; finally, $d_{-1}$ is the complex conjugate of $d_1$ above, hence $d_{-1}=0$ also.
        \end{itemize}
        As a result, in the ordered basis $\psi_0,\psi_1,\psi_{-1}$, the transformation $\psi\mapsto \cos(\theta)\psi$ has the following matrix (when restricting to the first $3\times 3$ entries):
        \begin{align}
            \begin{pmatrix}
                0 & \frac{1}{2} & \frac{1}{2}\\
                \frac{1}{2} & 0 & 0\\
                \frac{1}{2} & 0 & 0\\
            \end{pmatrix}
        \end{align}
        Then, with $H' = Eqr\cos(\theta)$, the above matrix is multiplied by $Eqr$, hence we get the following:
        \begin{align}
            H' \longrightarrow \frac{Eqr}{2}\begin{pmatrix}
                0 & 1 & 1\\
                1 & 0 & 0\\
                1 & 0 & 0\\
            \end{pmatrix}
        \end{align}
        Hence, the selection rules states: the state cannot transform to any state with same energy (EX: $\psi_0$ can't transform to $\psi_0$, $\psi_{\pm 1}$ can't transform to $\psi_{\pm 1}$). So, the concise form is $n-m = \pm 1$, if it transforms from $n$ to $m$.

        \hfil

        \item[(d)] Based on the $H'$'s matrix, if it's initially in ground state, it can transform from $\psi_0$ to $\psi_{\pm 1}$ (from ground state to first excited states), hence the energy it can absorb is $E_1-E_0 = \frac{\hbar^2}{2mr^2}$, so with each photon having energy $\hbar w$, the frequency it can absorb is $w=\frac{E_1-E_0}{\hbar}=\frac{\hbar}{2mr^2}$.
        
        \hfil

        \item[(e)] If consider up to $n=\pm 2$, notice that one of the first excited state $\psi_{-1}$ after transformation, say $\frac{1}{\sqrt{2\pi}}\cos(\theta)e^{-i\theta}$, its coefficient $d_{-2}$ (corresponding to $n=-2$ state $\psi_{-2}=\frac{1}{\sqrt{2\pi}}e^{-i2\theta}$) is given as follow:
        \begin{align}
            d_2 = \int_{0}^{2\pi}\frac{1}{\sqrt{2\pi}}\cos(\theta)e^{-i\theta}\frac{1}{\sqrt{2\pi}}e^{i2\theta}d\theta = \frac{1}{2\pi}\int_{0}^{2\pi}\cos(\theta)e^{i\theta}d\theta=\frac{1}{2}
        \end{align}
        Hence, it's also possible to jump from energ level $E_1$ to $E_2$. Then, the frequency it can absorb also includes:
        \begin{align}
            w' = \frac{E_2-E_1}{\hbar} = \frac{3\hbar}{2mr^2}
        \end{align}
    \end{itemize}
\end{proof}
